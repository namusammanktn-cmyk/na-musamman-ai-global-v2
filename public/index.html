async function sendMessage() { const text = messageInput.value.trim(); if (!text && !selectedImage) { return; } const imageToSend = selectedImage; addMessage( "user", text, imageToSend ); conversation.push({ role: "user", content: text || "Image uploaded" }); messageInput.value = ""; removeImage.click(); sendBtn.disabled = true; sendBtn.textContent = "…"; try { const response = await fetch( "/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: text, image: imageToSend, history: conversation.slice(-10) }) } ); const data = await response.json(); if (!response.ok) { throw new Error( data.error || "AI connection error." ); } const answer = data.answer || "No answer received."; addMessage( "assistant", answer ); conversation.push({ role: "assistant", content: answer }); saveHistory(); renderHistory(); } catch (error) { addMessage( "assistant", error.message || "AI connection error. Please try again." ); } finally { sendBtn.disabled = false; sendBtn.textContent = "➤"; } }
